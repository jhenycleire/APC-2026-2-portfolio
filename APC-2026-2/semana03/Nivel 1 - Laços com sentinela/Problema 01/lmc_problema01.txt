loop    INP          // Lê o número digitado e coloca no accumulator
	BRZ fim      // Decisão, se for igual a zero o programa acaba
        ADD soma     // Soma o valor que está no accumulator com o valor da varíavel soma
        STA soma     // Salva o novo resultado na varíavel soma
        BRA loop     // Retoma ao início para que outro número seja digitado
fim     LDA soma     // Carrega o valor total da soma no accumulator 
	OUT          // Mostra o valor da soma na tela
	HLT	     // Encerra o programa
		
soma    DAT 0        // A varíavel soma se inicia igual a zero
